The line was written as <code>tr … &lt; /proc/$pid/environ 2&gt;/dev/null</code>, which looks safe and is not: the shell applies redirections left to right, so the input redirect fails and prints its error while <code>stderr</code> is still the terminal. The suppression arrives one step too late. Fixed by testing the file is readable first.</p></div>

      <div class="learn"><span class="dot"></span><p><b>A cleanup command killed the session running it.</b> <code>pkill -f 't3code'</code> matches against full command lines — including the command line of the shell that is running <code>pkill</code>, because the pattern appears in it. It killed the app, then killed itself, and the whole step exited with code 144 halfway through. The fix is the old bracket trick: write the pattern as <code>[t]3code</code>, which still matches the target but no longer matches the literal text sitting in the shell's own command line.</p></div>

def scrub(s, safe):
    if not safe:
        return s
    for pat, rep in REDACTIONS:
        s = pat.sub(rep, s)
    return s


def load(path):
    out = []
    for line in open(path):
        line = line.strip()
        if not line:
            continue
        try:
            out.append(json.loads(line))
        except Exception:
            pass
    return out


def blocks(rows, safe):
    """Yield (kind, header, body) tuples in order."""
    pending_tool = {}
    for r in rows:
        t = r.get('type')
        ts = (r.get('timestamp') or '')[11:16]
        m = r.get('message')
        if not isinstance(m, dict):
            continue
        c = m.get('content')

        if t == 'user' and isinstance(c, str):
            yield ('user', ts, scrub(c, safe))

        elif t == 'user' and isinstance(c, list):
            for b in c:
                if not isinstance(b, dict) or b.get('type') != 'tool_result':
                    continue
                cont = b.get('content')
                if isinstance(cont, list):
                    cont = '\n'.join(x.get('text', '') for x in cont if isinstance(x, dict))
                cont = cont if isinstance(cont, str) else json.dumps(cont)[:400]
                name, cmd = pending_tool.get(b.get('tool_use_id'), ('tool', ''))
                if safe and any(d in cmd for d in DROP_RESULT_IF):
                    cont = DROP_NOTE
                else:
                    cont = scrub(cont, safe)
                    if len(cont) > MAX_RESULT:
                        cont = cont[:MAX_RESULT] + f'\n… [{len(cont) - MAX_RESULT} more characters]'
                yield ('result', name, cont)

        elif t == 'assistant' and isinstance(c, list):
            for b in c:
                if not isinstance(b, dict):
                    continue
                if b.get('type') == 'text' and b.get('text', '').strip():
                    yield ('assistant', ts, scrub(b['text'][:MAX_TEXT], safe))
                elif b.get('type') == 'tool_use':
                    inp = b.get('input') or {}
                    name = b.get('name', 'tool')
                    cmd = inp.get('command') or inp.get('file_path') or inp.get('prompt') or ''
                    if not isinstance(cmd, str):
                        cmd = json.dumps(cmd)[:800]
                    pending_tool[b.get('id')] = (name, cmd)
                    desc = inp.get('description', '')
                    yield ('tool', f'{name}|{desc}', scrub(cmd, safe))


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('jsonl')
    ap.add_argument('--safe', action='store_true')
    ap.add_argument('--format', choices=['txt', 'html'], default='txt')
    a = ap.parse_args()

    rows = load(a.jsonl)
    bs = list(blocks(rows, a.safe))

    if a.format == 'txt':
        for kind, hdr, body in bs:
            if kind == 'user':
                print(f'\n{"="*70}\nUSER  {hdr}\n{"="*70}\n{body}')
            elif kind == 'assistant':
                print(f'\nCLAUDE  {hdr}\n{"-"*30}\n{body}')
            elif kind == 'tool':
                name, desc = hdr.split('|', 1)
                print(f'\n  [{name}] {desc}\n  $ {body}')
            else:
                print(f'  → {body}\n')
        return

    # html fragment
    parts = []
    for kind, hdr, body in bs:
        e = html.escape(body)
        if kind == 'user':
            parts.append(f'<div class="exhibit"><div class="ex-head">you <span class="lang">{hdr}</span></div>'
                         f'<pre class="out">{e}</pre></div>')
        elif kind == 'assistant':
            parts.append(f'<div class="exhibit"><div class="ex-head">claude <span class="lang">{hdr}</span></div>'
                         f'<pre class="out">{e}</pre></div>')
        elif kind == 'tool':
            name, desc = hdr.split('|', 1)
            parts.append(f'<div class="exhibit"><div class="ex-head">{html.escape(name)}'
                         f'<span class="lang">{html.escape(desc)}</span></div>'
                         f'<pre class="out"><span class="bar">$</span> {e}</pre></div>')
        else:
            parts.append(f'<div class="exhibit"><div class="ex-head">result '
                         f'<span class="lang">{html.escape(hdr)}</span></div>'
                         f'<pre class="out"><span class="dim">{e}</span></pre></div>')
    print('\n'.join(parts))


if __name__ == '__main__':
    main()
